INSERT INTO wrestling_nchsaa_results (year, classification, weight_class, place, wrestler_name, school)
SELECT 2021, '4A', '106', 1, 'Cam Stinson', 'Mallard Creek'
WHERE NOT EXISTS (
  SELECT 1 FROM wrestling_nchsaa_results
  WHERE year = 2021
    AND (wrestler_name ILIKE '%Cam%Stinson%' OR wrestler_name ILIKE '%Cameron%Stinson%')
    AND place = 1
);
