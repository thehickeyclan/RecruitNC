-- 1) Ensure paused status is allowed (skip if table was created from blue-wiq-subscriptions.sql.txt)
alter table public.blue_wiq_subscriptions drop constraint if exists blue_wiq_subscriptions_status_check;
alter table public.blue_wiq_subscriptions add constraint blue_wiq_subscriptions_status_check
  check (status in ('active', 'past_due', 'grace', 'cancelled', 'paused'));

-- 2) Normalize names like lib/blue-wiq-import.ts (lowercase, hyphens → spaces, collapse whitespace)
create or replace function public._wiq_norm_name(t text)
returns text
language sql
immutable
as $$
  select lower(regexp_replace(regexp_replace(trim(coalesce(t, '')), '-', ' ', 'g'), '\s+', ' ', 'g'));
$$;

-- Wrestlers with exactly one allowlist row (fallback match by wrestler only)
drop table if exists _wiq_active_june_wrestler_singleton;
create temporary table _wiq_active_june_wrestler_singleton as
select public._wiq_norm_name(wrestler_name) as wrestler_norm
from _wiq_active_june_2026
group by public._wiq_norm_name(wrestler_name)
having count(*) = 1;

-- 4) PREVIEW — rows that will stay active
select
  s.wrestler_name,
  s.billed_to,
  s.wiq_billing_partner_id,
  s.status
from public.blue_wiq_subscriptions s
where s.status = 'active'
  and (
    exists (
      select 1
      from _wiq_active_june_2026 a
      where public._wiq_norm_name(a.wrestler_name) = public._wiq_norm_name(s.wrestler_name)
        and public._wiq_norm_name(a.billed_to) = public._wiq_norm_name(s.billed_to)
    )
    or exists (
      select 1
      from _wiq_active_june_wrestler_singleton sg
      where sg.wrestler_norm = public._wiq_norm_name(s.wrestler_name)
    )
  )
order by s.wrestler_name, s.billed_to;

-- 5) PREVIEW — rows that will be demoted to paused
select
  s.wrestler_name,
  s.billed_to,
  s.wiq_billing_partner_id,
  s.status
from public.blue_wiq_subscriptions s
where s.status = 'active'
  and not (
    exists (
      select 1
      from _wiq_active_june_2026 a
      where public._wiq_norm_name(a.wrestler_name) = public._wiq_norm_name(s.wrestler_name)
        and public._wiq_norm_name(a.billed_to) = public._wiq_norm_name(s.billed_to)
    )
    or exists (
      select 1
      from _wiq_active_june_wrestler_singleton sg
      where sg.wrestler_norm = public._wiq_norm_name(s.wrestler_name)
    )
  )
order by s.wrestler_name, s.billed_to;

-- 6) APPLY — demote false actives
update public.blue_wiq_subscriptions s
set
  status = 'paused',
  wiq_status_raw = 'Not on active renewing report',
  updated_at = now()
where s.status = 'active'
  and not (
    exists (
      select 1
      from _wiq_active_june_2026 a
      where public._wiq_norm_name(a.wrestler_name) = public._wiq_norm_name(s.wrestler_name)
        and public._wiq_norm_name(a.billed_to) = public._wiq_norm_name(s.billed_to)
    )
    or exists (
      select 1
      from _wiq_active_june_wrestler_singleton sg
      where sg.wrestler_norm = public._wiq_norm_name(s.wrestler_name)
    )
  );

-- 7) VERIFY counts
select status, count(*) as n
from public.blue_wiq_subscriptions
group by status
order by status;

select count(*) as active_should_be_near_48
from public.blue_wiq_subscriptions
where status = 'active';
