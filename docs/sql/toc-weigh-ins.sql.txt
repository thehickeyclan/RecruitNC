-- TOC Friday weigh-in check-in.
-- One row per wrestler; the scale stations and the head table all read and write this table
-- through /api/admin/toc/weigh-ins using the service role, so RLS denies everything else.

create table if not exists public.toc_weigh_ins (
  athlete_id        uuid primary key references public.athletes(id) on delete cascade,
  weight_class      integer not null,
  recorded_weight   numeric(5,1),
  skin_check        text check (skin_check in ('pass', 'fail')),
  lanyard_given     boolean not null default false,
  notes             text,
  recorded_by       uuid,
  recorded_by_name  text,
  created_at        timestamptz not null default now(),
  updated_at        timestamptz not null default now()
);

create index if not exists idx_toc_weigh_ins_weight on public.toc_weigh_ins (weight_class);

alter table public.toc_weigh_ins enable row level security;

drop policy if exists toc_weigh_ins_deny_all on public.toc_weigh_ins;
create policy toc_weigh_ins_deny_all on public.toc_weigh_ins for all using (false);
