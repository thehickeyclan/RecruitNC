create or replace function public._wiq_norm_name(t text)
returns text
language sql
immutable
as $$
  select lower(regexp_replace(regexp_replace(trim(coalesce(t, '')), '-', ' ', 'g'), '\s+', ' ', 'g'));
$$;

create or replace function public._wiq_norm_billed(t text)
returns text
language sql
immutable
as $$
  select regexp_replace(public._wiq_norm_name(t), '\s+family$', '', 'g');
$$;

-- A) Allowlist slots with no DB row (strict match)
select
  a.wrestler_name,
  a.billed_to,
  'no_db_row_strict' as issue
from _wiq_active_june_2026 a
where not exists (
  select 1
  from public.blue_wiq_subscriptions s
  where public._wiq_norm_name(s.wrestler_name) = public._wiq_norm_name(a.wrestler_name)
    and public._wiq_norm_billed(s.billed_to) = public._wiq_norm_billed(a.billed_to)
);

-- B) Allowlist match exists but status is paused (wrongly demoted — check spelling)
select
  a.wrestler_name as allow_wrestler,
  a.billed_to as allow_billed,
  s.wrestler_name as db_wrestler,
  s.billed_to as db_billed,
  s.status,
  s.wiq_billing_partner_id
from _wiq_active_june_2026 a
join public.blue_wiq_subscriptions s
  on public._wiq_norm_name(s.wrestler_name) = public._wiq_norm_name(a.wrestler_name)
  and public._wiq_norm_billed(s.billed_to) = public._wiq_norm_billed(a.billed_to)
where s.status = 'paused';

-- C) Wrestler on allowlist, close name in DB but billed_to differs (likely the 4)
select
  a.wrestler_name as allow_wrestler,
  a.billed_to as allow_billed,
  s.wrestler_name as db_wrestler,
  s.billed_to as db_billed,
  s.status,
  s.wiq_billing_partner_id
from _wiq_active_june_2026 a
join public.blue_wiq_subscriptions s
  on public._wiq_norm_name(s.wrestler_name) = public._wiq_norm_name(a.wrestler_name)
where public._wiq_norm_billed(s.billed_to) <> public._wiq_norm_billed(a.billed_to)
order by a.wrestler_name, s.billed_to;

-- D) Paused rows that look like allowlist wrestlers (fuzzy — find the 4)
select
  s.wrestler_name,
  s.billed_to,
  s.status,
  s.wiq_status_raw,
  s.wiq_billing_partner_id
from public.blue_wiq_subscriptions s
where s.status = 'paused'
  and exists (
    select 1
    from _wiq_active_june_2026 a
    where public._wiq_norm_name(a.wrestler_name) = public._wiq_norm_name(s.wrestler_name)
  )
order by s.wrestler_name;

-- E) Summary
select
  (select count(*) from public.blue_wiq_subscriptions where status = 'active') as active_now,
  (select count(*) from _wiq_active_june_2026) as allowlist_rows,
  (
    select count(distinct (public._wiq_norm_name(wrestler_name), public._wiq_norm_billed(billed_to)))
    from _wiq_active_june_2026
  ) as allowlist_unique_pairs;
