-- Tournament of Champions bracket pool.
--
-- Two tables: what entrants picked, and what actually happened. Scoring joins them in the app
-- (lib/toc/pool-scoring.ts) rather than in SQL, so the rules are unit-tested rather than trusted.
--
-- Run this whole file in the Supabase SQL editor.

-- ---------------------------------------------------------------------------
-- Entries: one row per user per weight class.
-- ---------------------------------------------------------------------------
create table if not exists public.toc_pool_entries (
  id            uuid primary key default gen_random_uuid(),
  user_id       uuid not null references auth.users (id) on delete cascade,
  weight_class  integer not null,

  -- { "<boutNumber>": "<athleteId>" }. Stored as given so a bracket that changes shape does not
  -- invalidate rows; scoring only reads bout numbers present in the official locked draw.
  picks         jsonb not null default '{}'::jsonb,

  -- Saved entries are drafts. Submitted entries are in the pool. Both stay editable until the
  -- deadline; the flag is what the entrant sees, not a lock.
  submitted     boolean not null default false,
  submitted_at  timestamptz,

  created_at    timestamptz not null default now(),
  updated_at    timestamptz not null default now(),

  constraint toc_pool_entries_user_weight_unique unique (user_id, weight_class)
);

create index if not exists toc_pool_entries_weight_idx on public.toc_pool_entries (weight_class);
create index if not exists toc_pool_entries_submitted_idx on public.toc_pool_entries (submitted) where submitted;

-- ---------------------------------------------------------------------------
-- Results: the official outcome of each bout, entered live during the tournament.
-- ---------------------------------------------------------------------------
create table if not exists public.toc_bout_results (
  weight_class      integer not null,
  bout_number       integer not null,
  winner_athlete_id text not null,
  -- Kept for display and for auditing a correction; scoring only needs the winner.
  loser_athlete_id  text,
  recorded_by       uuid references auth.users (id),
  recorded_at       timestamptz not null default now(),
  updated_at        timestamptz not null default now(),

  primary key (weight_class, bout_number)
);

-- ---------------------------------------------------------------------------
-- Row level security.
--
-- The pool's promise is that nobody sees anyone else's picks. Entries are readable only by the
-- person who made them; the leaderboard is served by an API route using the service role, which
-- returns names and totals and never picks.
-- ---------------------------------------------------------------------------
alter table public.toc_pool_entries enable row level security;
alter table public.toc_bout_results enable row level security;

drop policy if exists "own entries readable" on public.toc_pool_entries;
create policy "own entries readable" on public.toc_pool_entries
  for select using (auth.uid() = user_id);

drop policy if exists "own entries writable" on public.toc_pool_entries;
create policy "own entries writable" on public.toc_pool_entries
  for all using (auth.uid() = user_id) with check (auth.uid() = user_id);

-- Results are public: they are what happened on the mat.
drop policy if exists "results readable by everyone" on public.toc_bout_results;
create policy "results readable by everyone" on public.toc_bout_results
  for select using (true);

-- Writes go through the service role only, so no client policy is granted.

-- ---------------------------------------------------------------------------
-- Keep updated_at honest.
-- ---------------------------------------------------------------------------
create or replace function public.toc_pool_touch_updated_at()
returns trigger language plpgsql as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

drop trigger if exists toc_pool_entries_touch on public.toc_pool_entries;
create trigger toc_pool_entries_touch before update on public.toc_pool_entries
  for each row execute function public.toc_pool_touch_updated_at();

drop trigger if exists toc_bout_results_touch on public.toc_bout_results;
create trigger toc_bout_results_touch before update on public.toc_bout_results
  for each row execute function public.toc_pool_touch_updated_at();

-- ---------------------------------------------------------------------------
-- Finals tiebreaker.
--
-- Total points tie constantly — early on, most entrants have the favourites and finish level.
-- The tiebreaker asks how the championship match ends: the method, plus a score when the method
-- has one worth guessing. A fall and a tech fall are their own answers.
--
-- Safe to re-run.
-- ---------------------------------------------------------------------------

alter table public.toc_pool_entries
  add column if not exists final_method       text,
  add column if not exists final_winner_score integer,
  add column if not exists final_loser_score  integer;

alter table public.toc_pool_entries
  drop constraint if exists toc_pool_entries_final_method_check;
alter table public.toc_pool_entries
  add constraint toc_pool_entries_final_method_check
  check (final_method is null or final_method in ('FALL','TF','MAJ','DEC'));

alter table public.toc_bout_results
  add column if not exists method       text,
  add column if not exists winner_score integer,
  add column if not exists loser_score  integer;

alter table public.toc_bout_results
  drop constraint if exists toc_bout_results_method_check;
alter table public.toc_bout_results
  add constraint toc_bout_results_method_check
  check (method is null or method in ('FALL','TF','MAJ','DEC'));
