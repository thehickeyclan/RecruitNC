-- Uses relaxed billed_to match (strips trailing " Family"). Safe to re-run.
-- Run blue-wiq-diagnose-active-gap.sql first if you want to preview.

create or replace function public._wiq_norm_name(t text)
returns text
language sql
immutable
as $$
  select lower(regexp_replace(regexp_replace(trim(coalesce(t, '')), '-', ' ', 'g'), '\s+', ' ', 'g'));
$$;

create or replace function public._wiq_norm_billed(t text)
returns text
language sql
immutable
as $$
  select regexp_replace(public._wiq_norm_name(t), '\s+family$', '', 'g');
$$;

-- PREVIEW: will be re-activated
select
  s.wrestler_name,
  s.billed_to,
  s.status,
  s.wiq_billing_partner_id
from public.blue_wiq_subscriptions s
where s.status in ('paused', 'past_due', 'grace')
  and exists (
    select 1
    from _wiq_active_june_2026 a
    where public._wiq_norm_name(a.wrestler_name) = public._wiq_norm_name(s.wrestler_name)
      and public._wiq_norm_billed(a.billed_to) = public._wiq_norm_billed(s.billed_to)
  )
order by s.wrestler_name;

-- APPLY: restore allowlist members (fixes Cameron Boltes Family / hyphen billed_to mismatches)
update public.blue_wiq_subscriptions s
set
  status = 'active',
  wiq_status_raw = 'Paid',
  updated_at = now()
where s.status in ('paused', 'past_due', 'grace')
  and exists (
    select 1
    from _wiq_active_june_2026 a
    where public._wiq_norm_name(a.wrestler_name) = public._wiq_norm_name(s.wrestler_name)
      and public._wiq_norm_billed(a.billed_to) = public._wiq_norm_billed(s.billed_to)
  );

-- Re-demote any active row NOT on allowlist (same relaxed match)
update public.blue_wiq_subscriptions s
set
  status = 'paused',
  wiq_status_raw = 'Not on active renewing report',
  updated_at = now()
where s.status = 'active'
  and not exists (
    select 1
    from _wiq_active_june_2026 a
    where public._wiq_norm_name(a.wrestler_name) = public._wiq_norm_name(s.wrestler_name)
      and public._wiq_norm_billed(a.billed_to) = public._wiq_norm_billed(s.billed_to)
  );

select count(*) as active_after_fix
from public.blue_wiq_subscriptions
where status = 'active';
