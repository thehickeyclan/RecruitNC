-- WrestlingIQ external subscription registry (billing stays in WIQ; RecruitNC tracks roster only).
-- Run in Supabase SQL Editor.

create table if not exists public.blue_wiq_subscriptions (
  id uuid primary key default gen_random_uuid(),
  wiq_billing_partner_id text not null unique,
  wrestler_name text not null,
  billed_to text,
  status text not null default 'active'
    check (status in ('active', 'past_due', 'grace', 'cancelled', 'paused')),
  wiq_status_raw text,
  member_since timestamptz,
  next_due_at timestamptz,
  active_until timestamptz,
  amount_cents integer,
  amount_display text,
  billing_interval text,
  membership_type text,
  product_label text not null default 'NC United Blue',
  discount_code text,
  athlete_id uuid,
  match_confidence text check (match_confidence in ('exact', 'fuzzy', 'manual', 'unmatched')),
  notes text,
  first_seen_at timestamptz not null default now(),
  last_seen_at timestamptz not null default now(),
  last_import_at timestamptz not null default now(),
  missing_from_last_import boolean not null default false,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create index if not exists idx_blue_wiq_subs_athlete on public.blue_wiq_subscriptions(athlete_id);
create index if not exists idx_blue_wiq_subs_status on public.blue_wiq_subscriptions(status);
create index if not exists idx_blue_wiq_subs_last_seen on public.blue_wiq_subscriptions(last_seen_at desc);

create table if not exists public.blue_wiq_import_runs (
  id uuid primary key default gen_random_uuid(),
  imported_at timestamptz not null default now(),
  imported_by uuid,
  file_label text,
  total_rows integer not null default 0,
  blue_rows integer not null default 0,
  upserted integer not null default 0,
  matched integer not null default 0,
  flagged_missing integer not null default 0,
  summary jsonb
);

alter table public.blue_wiq_subscriptions enable row level security;
alter table public.blue_wiq_import_runs enable row level security;

drop policy if exists "Service role full access blue_wiq_subscriptions" on public.blue_wiq_subscriptions;
create policy "Service role full access blue_wiq_subscriptions"
  on public.blue_wiq_subscriptions for all to service_role using (true) with check (true);

drop policy if exists "Service role full access blue_wiq_import_runs" on public.blue_wiq_import_runs;
create policy "Service role full access blue_wiq_import_runs"
  on public.blue_wiq_import_runs for all to service_role using (true) with check (true);
