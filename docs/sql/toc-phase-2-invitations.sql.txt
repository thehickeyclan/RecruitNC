-- Tournament of Champions Phase 2 — athlete confirmation columns
-- Run in Supabase SQL Editor after toc-phase-1.sql.txt

alter table public.toc_invitations
  add column if not exists jacket_size text,
  add column if not exists medical_notes text,
  add column if not exists photo_release_accepted boolean default false,
  add column if not exists weight_acknowledgment boolean default false,
  add column if not exists usaw_acknowledgment boolean default false,
  add column if not exists attendance_acknowledgment boolean default false;

create unique index if not exists idx_toc_invitations_athlete_unique
  on public.toc_invitations (athlete_id);

create index if not exists idx_toc_invitations_athlete_status
  on public.toc_invitations (athlete_id, status);
